cmake_minimum_required(VERSION 3.24)
project(VkEngine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
find_package(Vulkan 1.3 REQUIRED COMPONENTS glslc)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(TINYGLTF_BUILD_LOADER_EXAMPLE OFF CACHE BOOL "" FORCE)
set(TINYGLTF_BUILD_GL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TINYGLTF_HEADER_ONLY ON CACHE BOOL "" FORCE)
set(TINYGLTF_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_Declare(tinygltf GIT_REPOSITORY https://github.com/syoyo/tinygltf.git GIT_TAG v2.9.5 GIT_SHALLOW TRUE)
FetchContent_Declare(assimp GIT_REPOSITORY https://github.com/assimp/assimp.git GIT_TAG v5.4.3 GIT_SHALLOW TRUE)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.92.9 GIT_SHALLOW TRUE)
FetchContent_Declare(imguizmo GIT_REPOSITORY https://github.com/CedricGuillemet/ImGuizmo.git GIT_TAG 18cef5e031d8c6973d80284c67f60549fafd78c1 SOURCE_SUBDIR unused)
FetchContent_Declare(mikktspace GIT_REPOSITORY https://github.com/mmikk/MikkTSpace.git GIT_TAG 3e895b49d05ea07e4c2133156cfa94369e19e409)
FetchContent_Declare(vma GIT_REPOSITORY https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator.git GIT_TAG v3.3.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw glm tinygltf assimp imgui imguizmo mikktspace vma)
add_library(vke_assets src/Asset.cpp src/Scene.cpp ${mikktspace_SOURCE_DIR}/mikktspace.c)
target_include_directories(vke_assets PUBLIC src ${tinygltf_SOURCE_DIR} PRIVATE ${mikktspace_SOURCE_DIR})
target_link_libraries(vke_assets PUBLIC glm::glm assimp)
target_compile_definitions(vke_assets PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_RADIANS NOMINMAX)
add_library(vke_ui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp ${imguizmo_SOURCE_DIR}/src/ImGuizmo.cpp)
target_include_directories(vke_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${imguizmo_SOURCE_DIR}/src)
target_link_libraries(vke_ui PUBLIC glfw Vulkan::Vulkan)
add_executable(VkEngine src/main.cpp src/Renderer.cpp src/Platform.cpp)
target_link_libraries(VkEngine PRIVATE vke_assets vke_ui GPUOpen::VulkanMemoryAllocator comdlg32 shell32)
target_compile_definitions(VkEngine PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
  foreach(t vke_assets vke_ui VkEngine)
    target_compile_options(${t} PRIVATE /utf-8 /EHsc /MP)
  endforeach()
endif()
file(GLOB shaders CONFIGURE_DEPENDS shaders/*.vert shaders/*.frag shaders/*.comp)
foreach(shader ${shaders})
  get_filename_component(name ${shader} NAME)
  set(out ${CMAKE_CURRENT_BINARY_DIR}/shaders/${name}.spv)
  add_custom_command(OUTPUT ${out} COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders COMMAND Vulkan::glslc --target-env=vulkan1.3 -O -I${CMAKE_CURRENT_SOURCE_DIR}/shaders ${shader} -o ${out} DEPENDS ${shader} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/common.glsl VERBATIM)
  list(APPEND compiled_shaders ${out})
endforeach()
add_custom_target(vke_shaders ALL DEPENDS ${compiled_shaders})
add_dependencies(VkEngine vke_shaders)
add_custom_command(TARGET VkEngine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders $<TARGET_FILE_DIR:VkEngine>/shaders)
enable_testing()
add_executable(vke_tests tests/tests.cpp)
target_link_libraries(vke_tests PRIVATE vke_assets)
target_compile_definitions(vke_tests PRIVATE VKE_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
if(MSVC)
  target_compile_options(vke_tests PRIVATE /utf-8 /EHsc)
endif()
add_test(NAME assets_scene COMMAND vke_tests)
install(TARGETS VkEngine RUNTIME DESTINATION .)
install(DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/shaders DESTINATION .)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES ${glfw_SOURCE_DIR}/LICENSE.md DESTINATION licenses RENAME GLFW.txt)
install(FILES ${glm_SOURCE_DIR}/copying.txt DESTINATION licenses RENAME GLM.txt)
install(FILES ${tinygltf_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME tinygltf.txt)
install(FILES ${tinygltf_SOURCE_DIR}/json.hpp ${tinygltf_SOURCE_DIR}/stb_image.h ${tinygltf_SOURCE_DIR}/stb_image_write.h DESTINATION licenses/source-notices)
install(FILES ${assimp_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME Assimp.txt)
install(FILES ${assimp_SOURCE_DIR}/contrib/zlib/LICENSE DESTINATION licenses RENAME zlib.txt)
install(FILES ${assimp_SOURCE_DIR}/contrib/openddlparser/LICENSE DESTINATION licenses RENAME OpenDDLParser.txt)
install(FILES ${assimp_SOURCE_DIR}/contrib/poly2tri/LICENSE DESTINATION licenses RENAME Poly2Tri.txt)
install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME ImGui.txt)
install(FILES ${imguizmo_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME ImGuizmo.txt)
install(FILES ${mikktspace_SOURCE_DIR}/mikktspace.h DESTINATION licenses/source-notices)
install(FILES ${vma_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME VMA.txt)
